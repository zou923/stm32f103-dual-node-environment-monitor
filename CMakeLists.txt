cmake_minimum_required(VERSION 3.20)
project(stm32f103c8t6 LANGUAGES C ASM)
set(STM32CUBE_F1_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/STM32CubeF1")
set(COMMON_SOURCES Core/Src/stm32f1xx_it.c Hardware/I2C/I2C1.c Hardware/OLED/OLED.c Hardware/OLED/OLED_Port.c Hardware/SHT30/SHT30.c Hardware/BH1750/BH1750.c Hardware/NRF24L01/NRF24L01.c Hardware/Buzzer/Buzzer.c "${STM32CUBE_F1_ROOT}/Drivers/CMSIS/Device/ST/STM32F1xx/Source/Templates/system_stm32f1xx.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_cortex.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_dma.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_flash.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_flash_ex.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_gpio.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_i2c.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_spi.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_rcc.c" "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Src/stm32f1xx_hal_rcc_ex.c" startup_stm32f103c8tx.s)
set(MCU_FLAGS -mcpu=cortex-m3 -mthumb)
function(add_firmware target main_source)
 add_executable(${target} ${main_source} ${COMMON_SOURCES})
 set_target_properties(${target} PROPERTIES SUFFIX ".elf")
 target_include_directories(${target} PRIVATE Core/Inc Hardware/I2C Hardware/OLED Hardware/SHT30 Hardware/BH1750 Hardware/NRF24L01 Hardware/Buzzer "${STM32CUBE_F1_ROOT}/Drivers/STM32F1xx_HAL_Driver/Inc" "${STM32CUBE_F1_ROOT}/Drivers/CMSIS/Device/ST/STM32F1xx/Include" "${STM32CUBE_F1_ROOT}/Drivers/CMSIS/Include")
 target_compile_definitions(${target} PRIVATE USE_HAL_DRIVER STM32F103xB)
 target_compile_options(${target} PRIVATE ${MCU_FLAGS} -ffunction-sections -fdata-sections -Wall -Wextra -Og -g3)
 target_link_options(${target} PRIVATE ${MCU_FLAGS} -T${CMAKE_CURRENT_SOURCE_DIR}/STM32F103C8TX_FLASH.ld -nostartfiles --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,-Map=${CMAKE_BINARY_DIR}/${target}.map)
 add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:${target}> ${CMAKE_BINARY_DIR}/${target}.hex COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:${target}> ${CMAKE_BINARY_DIR}/${target}.bin COMMAND ${CMAKE_SIZE} $<TARGET_FILE:${target}>)
endfunction()
add_firmware(stm32f103-master Core/Src/main_master.c)
add_firmware(stm32f103-slave Core/Src/main_slave.c)
